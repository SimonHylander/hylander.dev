```ts twoslash title="retrieve.ts"
import { Data, Effect } from "effect";

class RetrievalError extends Data.TaggedError("RetrievalError")<{
  readonly query: string;
}> {}

type Chunk = { readonly text: string; readonly score: number };

declare const searchVectors: (
  query: string,
) => Effect.Effect<ReadonlyArray<Chunk>, RetrievalError>;

const retrieve = (query: string) =>
  //  ^?
  Effect.gen(function* () {
    const chunks = yield* searchVectors(query);
    return chunks.filter((chunk) => chunk.score > 0.75);
  });
```
